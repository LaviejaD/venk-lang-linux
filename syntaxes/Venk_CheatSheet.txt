1. BASIC SYNTAX
--------------------------------------------------------------------------------
// Variables
let x = 10;           // Immutable variable
mut y = 20;           // Mutable variable
let z: float = 3.14;  // With explicit type

// Loops & Conditions
if x > 10 { print("Big"); } elif x == 10 { print("Equal"); } else { print("Small"); }
while x > 0 { x -= 1; }
for i in range(0, 10) { print(i); }

// Functions & Structs
fn add(a: int, b: int) -> int { return a + b; }
struct Player { name: string, hp: int }
impl Player { fn heal(mut self, amount: int) { self.hp += amount; } }
